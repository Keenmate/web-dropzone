VU01 · What you need to tell Vue about custom elements

Vue treats unknown tags as components by default and warns when it can't resolve them. Tell the compiler that web-dropzone* tags are custom elements via app.config.compilerOptions.isCustomElement, and they pass through as native DOM. From there it's plain Vue reactivity: a ref for the store element, a ref for the file array, and a computed for whatever derived state the UI needs.

Pattern
  • isCustomElement: (tag) => tag.startsWith('web-dropzone') on the app config silences false-positive warnings.
  • ref<HTMLElement> + onMounted wires the listener — Vue's v-on shortcut doesn't bind to files-change on a child component because it isn't part of Vue's emit graph.
  • computed drives the stats grid; v-for with a :key keyed by file.id avoids row churn.

VU02 · Live demo

The Vue app below renders the picker, the file list, and a stats grid that updates reactively on every coalesced event tick. Pause / resume / retry buttons call directly into the store's imperative API.

VU03 · The whole component

import { createApp, ref, computed, onMounted, onBeforeUnmount } from 'https://esm.sh/vue@3/dist/vue.esm-browser.js'

const App = {
    setup() {
        const storeRef = ref(null)
        const files = ref([])

        const stats = computed(() => {
            const acc = { total: files.value.length, pending: 0, uploading: 0, complete: 0, error: 0 }
            for (const f of files.value) acc[f.status] = (acc[f.status] ?? 0) + 1
            return acc
        })

        const onChange = (e) => { files.value = e.detail.files }

        onMounted(() => {
            const s = storeRef.value
            s.uploadFileCallback = simulateUpload({ durationMs: 4000 })
            s.addEventListener('files-change', onChange)
        })

        onBeforeUnmount(() => {
            storeRef.value?.removeEventListener('files-change', onChange)
        })

        const rowAction = (f) => {
            const s = storeRef.value
            if (f.status === 'uploading') return s.pauseFile(f.id)
            if (f.status === 'paused')    return s.resumeFile(f.id)
            if (f.status === 'error')     return s.retryFile(f.id)
        }

        return { storeRef, files, stats, rowAction, iconFor, actionLabel }
    },
    template: `
        <web-dropzone ref="storeRef" id="vu-store" render-mode="headless" multiple max-file-count="6"></web-dropzone>
        <web-dropzone-picker for="vu-store" selector-appearance="card" :labels="{ prompt: 'Drop files for Vue' }"></web-dropzone-picker>
        <div v-if="files.length === 0" class="vu-empty">No files yet</div>
        <div v-else class="vu-list">
            <div v-for="f in files" :key="f.id" class="vu-row">
                <span>{{ iconFor(f.status) }}</span>
                <span class="vu-row__name">{{ f.name }}</span>
                <span :class="['vu-row__status', 'vu-row__status--' + f.status]">{{ f.status }}</span>
                <span class="vu-row__bar"><span class="vu-row__bar-fill" :style="{ width: f.progress + '%' }"></span></span>
                <button v-if="actionLabel(f.status)" class="vu-row__action" @click="rowAction(f)">{{ actionLabel(f.status) }}</button>
            </div>
        </div>
    `
}

const app = createApp(App)
app.config.compilerOptions.isCustomElement = (tag) => tag.startsWith('web-dropzone')
app.mount('#vue-root')
