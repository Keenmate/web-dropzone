render*Callback results: string = HTML, Node = mounted, null / undefined = default, false = hidden

Targeted by e2e/render-slots.spec.ts. Each callback reads its result from window.__mode[id].

#items — renderFileItemCallback per file: "def*" → null (built-in row), "skip*" → false (no row), else a string
#prompt — renderPromptCallback; #wrap — renderListWrapperCallback
#summary — renderSummaryCallback (card + popover)
#roll — the store's rolling list (structural); #roll-store + #roll-list — the satellite's (same shared renderer)
#ind-store + #ind — indicator renderFileInfoCallback
